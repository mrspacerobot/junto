# Map Panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the decorative SVG map in ExploreView with a real interactive Leaflet map showing activity pins from live API data.

**Architecture:** Leaflet map with `L.divIcon` markers and Leaflet popups in a single `MapPanel.vue`. Icon SVGs come from a build-time codegen step: a config file (`icon-names.json`) maps backend `iconName` values to Lucide SVG filenames, the codegen script reads those SVGs from `lucide-static` and generates a TypeScript map. No backend needed at build time. Backend `ActivityListItemResponse` gets `locationLat`/`locationLng` and `activityTypeIconName` so the map has everything from the list endpoint alone.

**Tech Stack:** Vue 3, Leaflet, lucide-static (build), Tailwind CSS

- [ ] **Step 2: Create the codegen script**

Create `frontend/scripts/generate-icon-map.ts`:

```ts
import { writeFileSync, readFileSync, mkdirSync } from 'fs'
import { join } from 'path'

// Read the config mapping: backend iconName → lucide SVG filename
const mapping: Record<string, string> = JSON.parse(
  readFileSync(join(__dirname, '../src/config/icon-names.json'), 'utf-8')
)

// Read each SVG from lucide-static, strip outer <svg> tag, keep inner elements
const entries = Object.entries(mapping).map(([iconName, lucideName]) => {
  const svgPath = join(__dirname, `../node_modules/lucide-static/icons/${lucideName}.svg`)
  try {
    const raw = readFileSync(svgPath, 'utf-8')
    const inner = raw
      .replace(/<svg[^>]*>/, '')
      .replace(/<\/svg>/, '')
      .trim()
    return { iconName, inner }
  } catch {
    console.warn(`⚠ Lucide icon "${lucideName}" not found (mapped from "${iconName}"), skipping`)
    return null
  }
}).filter(Boolean)

// Generate TypeScript file
const output = `// AUTO-GENERATED by scripts/generate-icon-map.ts — do not edit manually
// Regenerate: npm run generate-icons

export const iconMap: Record<string, string> = {
${entries.map(e => `  '${e!.iconName}': '${e!.inner.replace(/'/g, "\\'")}',`).join('\n')}
}

export const DEFAULT_ICON = '<circle cx="12" cy="12" r="10"/>'
`

mkdirSync(join(__dirname, '../src/generated'), { recursive: true })
writeFileSync(join(__dirname, '../src/generated/icon-map.ts'), output)
console.log(`✓ Generated icon map with ${entries.length} icons`)
```

### Task 4: Rewrite MapPanel.vue with Leaflet

This covers the full map: markers with codegen icons, popups, edge cases, and XSS protection.

**Files:**
- Rewrite: `frontend/src/components/map/MapPanel.vue`

- [ ] **Step 1: Write the complete MapPanel component**

```vue
<template>
  <div class="w-full h-full rounded-xl overflow-hidden relative">
    <div v-if="hasCoords" ref="mapContainer" class="w-full h-full"></div>
    <div
      v-else
      class="w-full h-full flex items-center justify-center"
      style="background: linear-gradient(135deg, #e8f0e0 0%, #d4e4c8 30%, #c2d6b4 60%, #dde8d4 100%);"
    >
      <p class="text-primary-600/60 text-sm font-medium">No activities with location</p>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch, onMounted, onBeforeUnmount, nextTick } from 'vue'
import { useRouter } from 'vue-router'
import L from 'leaflet'
import 'leaflet/dist/leaflet.css'
import { iconMap, DEFAULT_ICON } from '@/generated/icon-map'

/** Default map center: Zürich, Switzerland. Override via center prop. */
const ZURICH_CENTER: [number, number] = [47.3769, 8.5417]

const props = defineProps<{
  activities: any[]
  center?: [number, number]
}>()

const router = useRouter()
const mapContainer = ref<HTMLDivElement>()
let map: L.Map | null = null
let markerLayer: L.LayerGroup | null = null

const hasCoords = computed(() =>
  props.activities.some(a => a.locationLat != null && a.locationLng != null)
)

function escapeHtml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;')
}

function escapeHtmlAttr(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
}

function createMarkerIcon(iconName?: string): L.DivIcon {
  const inner = iconName
    ? (iconMap[iconName.toLowerCase()] || DEFAULT_ICON)
    : DEFAULT_ICON

  return L.divIcon({
    className: 'custom-marker',
    html: `<div style="width:30px;height:30px;border-radius:50%;background:#3d5a1e;display:flex;align-items:center;justify-content:center;border:2px solid white;box-shadow:0 2px 8px rgba(0,0,0,0.2);cursor:pointer;">
      <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="white" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">${inner}</svg>
    </div>`,
    iconSize: [30, 30],
    iconAnchor: [15, 15],
    popupAnchor: [0, -18],
  })
}

function createPopupContent(activity: any): string {
  const title = escapeHtml(activity.title || 'Untitled')
  const city = escapeHtml(activity.city || 'TBD')
  const id = escapeHtmlAttr(activity.id || '')
  const date = activity.startsAt
    ? new Date(activity.startsAt).toLocaleDateString(undefined, {
        weekday: 'short', day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit',
      })
    : ''

  return `<div class="map-popup" data-activity-id="${id}" style="font-family:'DM Sans',sans-serif;cursor:pointer;">
    <div style="height:50px;background:linear-gradient(135deg,#d4e4c8,#c2d6b4);border-radius:10px 10px 0 0;display:flex;align-items:flex-end;padding:6px 10px;">
      <span style="font-size:10px;color:#3d5a1e;font-weight:600;">${escapeHtml(date)}</span>
    </div>
    <div style="padding:8px 10px 10px;">
      <div style="font-size:13px;font-weight:700;color:#1a1a1a;line-height:1.3;">${title}</div>
      <div style="font-size:11px;color:#6A6560;margin-top:2px;">${activity.currentParticipants || 0} joined · ${city}</div>
    </div>
  </div>`
}

function initMap() {
  if (!mapContainer.value) return

  const center = props.center || ZURICH_CENTER

  map = L.map(mapContainer.value, {
    center,
    zoom: 12,
    zoomControl: false,
    attributionControl: true,
  })

  L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
    attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> &copy; <a href="https://carto.com/">CARTO</a>',
    subdomains: 'abcd',
    maxZoom: 19,
  }).addTo(map)

  L.control.zoom({ position: 'topright' }).addTo(map)
  map.attributionControl.setPrefix('')

  markerLayer = L.layerGroup().addTo(map)

  // Navigate to activity detail when popup content is clicked.
  // Scoped to each popup's DOM element via Leaflet's popupopen/popupremove events.
  map.on('popupopen', (e: L.PopupEvent) => {
    const popup = e.popup
    const popupEl = popup.getElement()
    if (!popupEl) return

    const clickHandler = (ev: Event) => {
      const target = (ev.target as HTMLElement).closest('.map-popup')
      if (!target) return
      const activityId = (target as HTMLElement).getAttribute('data-activity-id')
      if (activityId) {
        router.push({ name: 'activity-detail', params: { id: activityId } })
      }
    }

    popupEl.addEventListener('click', clickHandler)
    popup.once('remove', () => {
      popupEl.removeEventListener('click', clickHandler)
    })
  })

  updateMarkers()
}

function updateMarkers() {
  if (!map || !markerLayer) return

  markerLayer.clearLayers()

  const activitiesWithCoords = props.activities.filter(
    a => a.locationLat != null && a.locationLng != null
  )

  if (activitiesWithCoords.length === 0) return

  const bounds = L.latLngBounds([])

  for (const activity of activitiesWithCoords) {
    const latlng = L.latLng(activity.locationLat, activity.locationLng)
    bounds.extend(latlng)

    const marker = L.marker(latlng, {
      icon: createMarkerIcon(activity.activityTypeIconName),
    })

    marker.bindPopup(createPopupContent(activity), {
      className: 'custom-popup',
      closeButton: false,
      minWidth: 200,
      maxWidth: 240,
    })

    markerLayer!.addLayer(marker)
  }

  map.fitBounds(bounds, { padding: [30, 30], maxZoom: 14 })
}

onMounted(() => {
  if (hasCoords.value) {
    nextTick(() => initMap())
  }
})

// Re-init map when coords appear/disappear (e.g. after first data fetch)
watch(hasCoords, (val) => {
  if (val && !map) {
    nextTick(() => initMap())
  } else if (!val && map) {
    map.remove()
    map = null
    markerLayer = null
  }
})

// Update markers when activities change (parent replaces the array on each refetch)
watch(() => props.activities, () => {
  nextTick(() => updateMarkers())
})

onBeforeUnmount(() => {
  map?.remove()
  map = null
  markerLayer = null
})
</script>
```
